drop database if exists company;
create database company;
use company;

create table employee (
  ssn      char(9)       not null,
  fname    varchar(30)   not null,
  lname    varchar(30)   not null,
  bdate    date,
  address  varchar(100),
  gender   char(1),
  salary   decimal(10,2),
  dno      int,
  primary key (ssn)
);

create table department (
  dnumber        int         not null,
  dname          varchar(50) not null,
  mgr_ssn        char(9),
  mgr_start_date date,
  primary key (dnumber)
);

create table project (
  pnumber   int         not null,
  pname     varchar(50) not null,
  plocation varchar(50),
  dno       int,
  primary key (pnumber)
);


alter table employee add constraint fk_emp_dept foreign key (dno) references department(dnumber);

alter table department add constraint fk_dept_mgr foreign key (mgr_ssn) references employee(ssn);

alter table project add constraint fk_proj_dept foreign key (dno) references department(dnumber);
